urls={

'exim':'https://www.exim.gov/open-government-data',

def run(item):
 k,u=item
 try:
  r=requests.get(u,timeout=16);open('/tmp/rel-src-'+k,'wb').write(r.content); print(k,r.status_code,len(r.content),r.url[:110],r.text[:65].replace('\n',' '),flush=True)
 except Exception as e:print(k,str(e)[:180],flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=12) as ex:list(ex.map(run,urls.items()))
